import tkinter as tk from tkinter import messagebox import random import pyttsx3 # Initialize Text-to-Speech engine engine = pyttsx3.init() class SubjectCallerApp: def __init__(self, root): self.root = root self.root.title("Voice Subject Caller") self.subjects = [] self.total_weight = 0.0 # --- UI Elements --- self.label_status = tk.Label(root, text="Remaining: 1.00", font=("Arial", 12, "bold")) self.label_status.pack(pady=10) # Input Frame frame = tk.Frame(root) frame.pack(pady=10) tk.Label(frame, text="Name:").grid(row=0, column=0) self.entry_name = tk.Entry(frame) self.entry_name.grid(row=0, column=1, padx=5) tk.Label(frame, text="Weight (0.01-1.0):").grid(row=1, column=0) self.entry_weight = tk.Entry(frame) self.entry_weight.grid(row=1, column=1, padx=5) # Buttons self.btn_add = tk.Button(root, text="Add Subject", command=self.add_subject, bg="#e1f5fe") self.btn_add.pack(pady=5) self.btn_start = tk.Button(root, text="START (Call Subject)", command=self.call_subject, state=tk.DISABLED, bg="#c8e6c9", font=("Arial", 10, "bold")) self.btn_start.pack(pady=10, fill=tk.X, padx=20) self.btn_reset = tk.Button(root, text="Reset", command=self.reset, bg="#ffcdd2") self.btn_reset.pack(pady=5) # Listbox to show subjects self.listbox = tk.Listbox(root, height=6) self.listbox.pack(pady=10, fill=tk.BOTH, padx=20) def add_subject(self): name = self.entry_name.get().strip() try: weight = round(float(self.entry_weight.get()), 2) remaining = round(1.0 - self.total_weight, 2) if name == "": messagebox.showwarning("Error", "Please enter a name.") elif weight > remaining: messagebox.showwarning("Error", f"Weight too high! Only {remaining} left.") elif weight <= 0: messagebox.showwarning("Error", "Weight must be higher than 0.") else: self.subjects.append((name, weight)) self.total_weight = round(self.total_weight + weight, 2) self.listbox.insert(tk.END, f"{name} - {int(weight*100)}%") # Update status new_remaining = round(1.0 - self.total_weight, 2) self.label_status.config(text=f"Remaining: {new_remaining}") # Clear inputs self.entry_name.delete(0, tk.END) self.entry_weight.delete(0, tk.END) # Enable Start if sum is 1.0 if self.total_weight >= 1.0: self.btn_start.config(state=tk.NORMAL) self.btn_add.config(state=tk.DISABLED) self.label_status.config(text="Total Reached: 1.0 (Ready!)", fg="green") except ValueError: messagebox.showerror("Error", "Please enter a valid number for weight.") def call_subject(self): names = [s[0] for s in self.subjects] weights = [s[1] for s in self.subjects] # Weighted random choice result = random.choices(names, weights=weights, k=1)[0] # Voice output engine.say(f"The selected subject is {result}") engine.runAndWait() messagebox.showinfo("Result", f"🎯 {result}") def reset(self): self.subjects = [] self.total_weight = 0.0 self.listbox.delete(0, tk.END) self.label_status.config(text="Remaining: 1.00", fg="black") self.btn_start.config(state=tk.DISABLED) self.btn_add.config(state=tk.NORMAL) if __name__ == "__main__": root = tk.Tk() root.geometry("300x450") app = SubjectCallerApp(root) root.mainloop()
